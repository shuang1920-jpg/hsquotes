                            console.log('匿名登录成功，openid:', openid);

                            userStatus.textContent = '✅ 已连接 (匿名)';

                            db = app.database();

                            isCloudReady = true;

                            // 加载数据

                            loadEntries();

                        } else {

                            userStatus.textContent = '❌ 登录失败';

                        }

                    }).catch(err => {

                        console.error('匿名登录失败:', err);

                        userStatus.textContent = '❌ 登录失败，请检查网络或控制台匿名登录是否开启';

                    });

                } catch (e) {

                    console.error('初始化云开发失败:', e);

                    userStatus.textContent = '❌ 初始化失败';

                }

            }


            // 如果页面加载时已经解锁，直接初始化

            if (sessionStorage.getItem('isUnlocked') === 'true') {

                // 等DOM完全加载

                if (document.readyState === 'complete') {

                    initCloudBase();

                } else {

                    window.addEventListener('load', initCloudBase);

                }

            } else {

                // 未解锁，等待解锁按钮触发

            }


            // ============================================================

            // 3. 数据库操作

            // ============================================================


            // 加载所有灵感（按日期降序）

            function loadEntries() {

                if (!db) return;

                const collection = db.collection(COLLECTION_NAME);

                // 按 date 降序，然后按 _id 降序（保证最新在前）

                collection.orderBy('date', 'desc').orderBy('_id', 'desc').get().then(res => {

                    const data = res.data || [];

                    renderEntries(data);

                }).catch(err => {

                    console.error('查询失败:', err);

                    entriesList.innerHTML = `<div class="empty-state"><span>⚠️</span>加载失败，请重试</div>`;

                });

            }


            // 渲染列表

            function renderEntries(entries) {

                countBadge.textContent = `${entries.length} 条`;


                if (entries.length === 0) {

                    entriesList.innerHTML = `

                        <div class="empty-state">

                            <span>🪐</span>

                            还没有灵感，写一条吧

                        </div>

                    `;

                    return;

                }


                let html = '';

                entries.forEach(entry => {

                    const parts = entry.date.split('-');

                    const displayDate = `${parts[0]}年${parts[1]}月${parts[2]}日`;

                    const safeText = entry.text.replace(/</g, '&lt;').replace(/>/g, '&gt;');

                    // 判断是否本人（通过 _openid 对比）

                    const isMine = (entry._openid === currentUser?.uid);

                    const deleteClass = isMine ? 'is-mine' : 'not-mine';

                    const deleteTitle = isMine ? '删除' : '仅可删除自己的';

                    html += `

                        <div class="entry-card" data-id="${entry._id}">

                            <div class="entry-text">${safeText}</div>

                            <div class="entry-footer">

                                <span class="entry-date">${displayDate}</span>

                                <button class="entry-delete ${deleteClass}" data-id="${entry._id}" title="${deleteTitle}" ${isMine ? '' : 'disabled'}>✕</button>

                            </div>

                        </div>

                    `;

                });


                entriesList.innerHTML = html;


                // 绑定删除事件（仅本人可删）

                entriesList.querySelectorAll('.entry-delete.is-mine').forEach(btn => {

                    btn.addEventListener('click', function(e) {

                        e.stopPropagation();

                        const id = this.getAttribute('data-id');

                        if (confirm('确定要删除这条灵感吗？')) {

                            deleteEntryById(id);

                        }

                    });

                });

            }


            // 添加新灵感

            function addEntry() {

                if (!isCloudReady) {

                    alert('云服务未就绪，请稍后');

                    return;

                }

                const text = ideaInput.value.trim();

                const date = dateInput.value;

                if (!text) {

                    ideaInput.style.borderColor = '#f87171';

                    setTimeout(() => ideaInput.style.borderColor = '', 600);

                    return;

                }

                if (!date) {

                    alert('请选择日期');

                    return;

                }


                saveBtn.disabled = true;

                saveBtn.textContent = '⏳ 存入中';


                const collection = db.collection(COLLECTION_NAME);

                collection.add({

                    text: text,

                    date: date,

                    // 系统会自动添加 _openid 字段（当前用户的匿名ID）

                }).then(res => {

                    // 成功

                    ideaInput.value = '';

                    setDefaultDate();

                    loadEntries(); // 刷新列表

                }).catch(err => {

                    console.error('添加失败:', err);

                    alert('添加失败，请重试');

                }).finally(() => {

                    saveBtn.disabled = false;

                    saveBtn.innerHTML = '✦ 存入';

                });

            }


            // 删除单条（仅本人可删，但后端已校验，此处前端只允许自己的按钮）

            function deleteEntryById(id) {

                if (!db) return;

                const collection = db.collection(COLLECTION_NAME);

                // 注意：权限为“读取全部，修改本人”，所以只有 _openid 匹配的文档才能被删除

                collection.doc(id).remove().then(() => {

                    loadEntries();

                }).catch(err => {

                    console.error('删除失败:', err);

                    alert('删除失败，可能您没有权限');

                });

            }


            // ============================================================

            // 4. 绑定UI事件

            // ============================================================

            saveBtn.addEventListener('click', addEntry);

            ideaInput.addEventListener('keydown', (e) => {

                if ((e.ctrlKey || e.metaKey) && e.key === 'Enter') {

                    e.preventDefault();

                    addEntry();

                }

            });


            // 如果云开发已经准备就绪，也可以手动刷新

            // 暴露 reload 到控制台方便调试

            window.reloadEntries = loadEntries;


        })();

    </script>
